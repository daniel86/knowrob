144 semweb::PropertyPtr Vocabulary::defineProperty_(const std::shared_ptr<Property> &p) {
145  definedProperties_[p->iri()] = p;
146  definedReifications_[p->reification()->iri()] = p;
147  // define the reification of the property as a concept which inherits knowrob:ReifiedRelation
148  // note further superclasses will be added through Property::addDirectParent
149  auto reification = p->reification();
150  definedClasses_[reification->iri()] = reification;
151  reification->addDirectParent(defineClass(reification::ReifiedRelation->stringForm()),
152  ImportHierarchy::ORIGIN_SYSTEM);
153  return p;
154 }

197 void Vocabulary::setFrequency(const std::string_view &iri, uint64_t frequency) {
198  if (frequency == 0) {
199  frequency_.erase(iri);
200  } else {
201  auto p_it = definedProperties_.find(iri);
202  if (p_it != definedProperties_.end()) {
203  frequency_[p_it->second->iri()] = frequency;
204  return;
205  }
206  auto c_it = definedClasses_.find(iri);
207  if (c_it != definedClasses_.end()) {
208  frequency_[c_it->second->iri()] = frequency;
209  return;
210  }
211  KB_WARN("cannot set frequency of unknown resource '{}'", iri);
212  }
213 }

230 void Vocabulary::addResourceType(const std::string_view &resource_iri, const std::string_view &type_iri) {
231  if (isObjectPropertyIRI(type_iri))
232  setPropertyFlag(resource_iri, OBJECT_PROPERTY);
233  else if (isDatatypePropertyIRI(type_iri))
234  setPropertyFlag(resource_iri, DATATYPE_PROPERTY);
235  else if (isAnnotationPropertyIRI(type_iri))
236  setPropertyFlag(resource_iri, ANNOTATION_PROPERTY);
237  else if (isReflexivePropertyIRI(type_iri))
238  setPropertyFlag(resource_iri, REFLEXIVE_PROPERTY);
239  else if (isTransitivePropertyIRI(type_iri))
240  setPropertyFlag(resource_iri, TRANSITIVE_PROPERTY);
241  else if (isSymmetricPropertyIRI(type_iri))
242  setPropertyFlag(resource_iri, SYMMETRIC_PROPERTY);
243  else if (isPropertyIRI(type_iri))
244  defineProperty(resource_iri);
245  else if (isClassIRI(type_iri))
246  defineClass(resource_iri);
